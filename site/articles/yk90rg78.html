#include<iostream>
#define MAXN 22
using namespace std;
long long f[MAXN][MAXN]={0};
int ctrl[MAXN][MAXN],n,m,hx,hy;
int d[9][2]={{0,0},{1,2},{1,-2},{-1,2},{-1,-2},{2,1},{2,-1},{-2,1},{-2,-1}};
int main(){
	cin>>n>>m>>hx>>hy;
	for(int i=0;i<9;i++){
		int tmpx=hx+d[i][0],tmpy=hy+d[i][1];
		if(tmpx>=0&&tmpx<=n&&tmpy>=0&&tmpy<=m)
			ctrl[tmpx][tmpy]=1;
	}
	for(int i=0;i<=n;i++)
		for(int j=0;j<=m;j++){
			if(ctrl[i][j])continue;
			if(i!=0)f[i][j]+=f[i-1][j];
			if(j!=0)f[i][j]+=f[i][j-1];
		}
		f[0][0]=1-ctrl[0][0];
		cout<<f[n][m];
	return 0;
}

#include <bits/stdc++.h>
using namespace std;
const int maxn=210;
long long dp[maxn][maxn];
int n,m,a,b;
bool vis[maxn][maxn];
const int x[]={0,-2,-1,1,2,2,1,-1,-2};
const int y[]={0,1,2,2,1,-1,-2,-2,-1};
int main(){
 memset(vis,true,sizeof(vis));
 cin>>n>>m>>a>>b;
 n+=2,m+=2,a+=2,b+=2;
 for(int i=1;i<=8;i++){ vis[a+x[i]][b+y[i]]=false; }
 vis[a][b]=false;
 dp[2][1]=1;
 for(int i=2;i<=n;i++){
  for(int j=2;j<=m;j++){
   if(vis[i][j]){ dp[i][j]=dp[i-1][j]+dp[i][j-1]; }
  }
 }
 cout<<dp[n][m]<<endl;
 return 0;
}

#include <bits/stdc++.h>
using namespace std;
int a,b,n,m;
long long f[23];
bool ma[23][23];
void work(long long x,long long y){
 ma[x][y]=true; ma[x-1][y-2]=true; ma[x-2][y-1]=true; ma[x-2][y+1]=true;
 ma[x-1][y+2]=true; ma[x+1][y-2]=true; ma[x+2][y-1]=true;
 ma[x+2][y+1]=true; ma[x+1][y+2]=true;
}
int main(){
 cin>>n>>m>>a>>b; a+=2; b+=2; n+=2; m+=2; work(a,b);
 f[2]=1;
 for(int i=2;i<=n;i++) for(int j=2;j<=m;j++){
  if(i==2&&j==2)continue;
  if(!ma[i][j])f[j]+=f[j-1]; else f[j]=0;
 }
 cout<<f[m];
 return 0;
}
